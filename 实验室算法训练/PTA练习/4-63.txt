#include<stdio.h>
void panduan (int n){
    int c=n;int a[10];
    int len=0,sum=0;
    int i=1;
    while (c/1!=0){
        
        a[i]=c%10;
        c/=10;
        i++;
        len++;}
    for (int i=1;i<=len;i++){
        int c=1;
        for (int j=1;j<=len;j++){
            c=c*a[i];
        }
        sum+=c;
    }
    if (sum==n){printf("%d\n",n);}
}
int main(){
    int n;
    scanf("%d",&n);
    int a=1;
    for (int i=0;i<n-1;i++){
        a*=10;
    }
    int b=a*10-1;
    for (int i=a;i<=b;i++){
        panduan(i);
    }
    return 0;
}

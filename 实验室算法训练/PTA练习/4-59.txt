#include<stdio.h>
#include<math.h>
int main(){
    int n,m;
    int a,b,s;
    char op;
    scanf("%d",&n);
    for (int i=0;i<n;i++){
         scanf("%d",&m);
        double cnt=0;double p=0;
        for (int j=0;j<m;j++){
            scanf("%d %c %d = %d",&a,&op,&b,&s);
            if (op=='+'){if (a+b==s) cnt++;}
            else if (op=='-'){if (a-b==s) cnt++;}
            else if (op=='*'){if (a*b==s) cnt++;}
            else if (op=='/'){ if (b != 0 && a/b==s) cnt++; }}
        p=100*cnt/(m*1.0);printf("%.2lf\n",p);
    }
    return 0;
}

#include<stdio.h>
int main(){
    int a[10000];
    int cnt=0;
    int i=0;
    while(scanf("%d",&a[i])==1){
        i++;cnt++;
    }
    for (int i=0;i<cnt-1;i++){
        for (int j=0;j<cnt-1-i;j++){
            if (a[j]<a[j+1]){
            int t=a[j+1];
            a[j+1]=a[j];
            a[j]=t;}
        }
    }
    int m=1;int c=0;
    for (int i=1;i<=cnt;i++){
        printf("%d - %d\n",m,a[i-1]);
        if (a[i-1]==a[i]){c++;continue;}
        else {m+=c+1;c=0;}
    }
    return 0;
}

#include<stdio.h>
int main(){
    int n,cnt=0,m;
    scanf("%d",&n);
    m=n*(n-1)*(n-2);
    for (int i=1;i<=n;i++){
        for (int j=1;j<=n;j++){
            if (i==j)continue;
            else {
                for (int q=1;q<=n;q++){
                    if (i==j || j==q || i==q)continue;
                    else if (cnt==0){printf("%d",i*100+j*10+q);cnt++;}
                    else {printf(" %d",i*100+j*10+q);cnt++;}
                    if (cnt==10){printf("\n");cnt=0;}
                }
            }
        }
    }
    if (cnt!=0){
    printf("\n%d",m);}
    else printf("%d",m);
    return 0;
}

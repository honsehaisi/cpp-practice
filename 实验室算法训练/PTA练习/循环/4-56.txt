#include<stdio.h>
int pd (int n){
    int nx,len=0,c;
    nx=n*n;c=n;
    while (c/1!=0){
        c/=10;
        len++;
    }
    int m=1;
    for (int i=0;i<len;i++){m*=10;}
    if (nx%m==n){printf("%d",n);return 1;}
    else return 0;
}
int main(){
    int x;
    scanf("%d",&x);
    while(1){
        if(pd(x)==1)break;
        x++;
    }
    return 0;
}

#include<stdio.h>
void zhuanhuan(int n){
    if (n == 0) { printf("0");}

    int bin[64], len = 0;
    while (n > 0) {
        bin[len++] = n % 2;
        n /= 2;
    }

    for (int i = len - 1; i >= 0; i--)
        printf("%d", bin[i]);
    printf("\n");
}

int main(){
    int x;
    while (scanf("%d",&x)==1){
        zhuanhuan(x);
    }
    return 0;
}

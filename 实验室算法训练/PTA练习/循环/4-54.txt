#include<stdio.h>
int main(){
    int n;
    scanf("%d",&n);
    while(n!=6174){
        int s[4],t=n;
        for (int i=0;i<4;i++){
            s[i]=t%10;
            t/=10;
        }
        for (int i=0;i<3;i++){
            for (int j=0;j<3-i;j++){
                if (s[j]>s[j+1]){
                    int tem=s[j];
                    s[j]=s[j+1];
                    s[j+1]=tem;}}
        }
        int max=s[0]+s[1]*10+s[2]*100+s[3]*1000;
        int min=s[0]*1000+s[1]*100+s[2]*10+s[3];
        n=max-min;
        printf("%d-%04d=%d\n",max,min,n);
    }
    return 0;
}

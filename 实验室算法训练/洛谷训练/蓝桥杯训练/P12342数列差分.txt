#include<stdio.h>
#include<string.h>
int sort (int *s,int len){
    for(int gap =len /2; gap>0;gap /=2)
    for (int i=gap;i<len;i++){
        int t=s[i];int j=i;
        while ( j>=gap && s[j-gap]>t ){
            s[j]=s[j-gap];
            j-=gap; 
        }
        s[j]=t;
    }
    return 0;
}
int a[100005],b[100005];
int main(){
    int n;scanf("%d",&n);
    for (int i=0;i<n;i++){
        scanf("%d",&a[i]);
    }
    for (int i=0;i<n;i++){
        scanf("%d",&b[i]);
    }
    sort(a,n);
    sort(b,n);
    int cnt=0;
    int i = 0, j = 0, keep = 0;
    while (i < n && j < n) {
    if (a[i] > b[j]) { keep++; i++; j++; }
    else             { i++; }
}
printf("%d\n", n - keep);
    return 0;
}

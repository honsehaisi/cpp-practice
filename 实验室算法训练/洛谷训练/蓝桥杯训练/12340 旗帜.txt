#include<stdio.h>
int main(){
    char m[]="LANQIAO";
    char s[105];
    int h,w;
    scanf("%d %d",&h,&w);
    int q=0;int cnt=0;int cnta=0;
    for (int i=0;i<h;i++){
        for (int j=0;j<w;j++){
            q=q%7;
            s[j]=m[q];
            if (s[j]=='A')cnta++;
            q++;
            if (q==7)q=0;
        }
        cnt++;
        q=0;
        q+=cnt;
    }
    printf("%d",cnta);
    return 0;
}

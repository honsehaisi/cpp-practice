#include<stdio.h>
int n,k,a[15];
void dfs (int dep){
    if (dep==n){
        for (int i=0;i<n;i++){
            printf("%d%c",a[i],(i==n-1? '\n':' '));
        }
        return;
    }
    for (int j=1;j<=k;j++){
        a[dep]=j;
        dfs(dep+1);
    }
}
int main(){scanf("%d %d",&n,&k);dfs(0);return 0;}

#include<stdio.h>
int n,k,a[15],used[15];
void dfs(int dep){
    if (dep==k){
        for (int i=0;i<k;i++){
            printf("%d%c",a[i],i==k-1?'\n':' ');
        }
        return;
    }
    for (int j=1;j<=n;j++){
        if (used[j]==1){continue;}
        used[j]=1;
        a[dep]=j;
        dfs(dep+1);
        used[j]=0;
    }
}
int main(){scanf("%d %d",&n,&k);dfs(0);return 0;}
